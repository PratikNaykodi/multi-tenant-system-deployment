    res.json({
        message: "Multi-Tenant API is running"

});

server.listen(PORT, () => {
    console.log(`Server running on http://localhost:${PORT}`);
});

23. AXIOS SERVICE
=================
File: frontend/src/services/api.js

import axios from "axios";

const api = axios.create({
    baseURL: import.meta.env.VITE_API_BASE_URL,
    headers: {
        "Content-Type": "application/json"
    }
});

api.interceptors.request.use(
    (config) => {
        const token = localStorage.getItem("token");
        const tenantId = localStorage.getItem("tenantId");

        if (token) {
            config.headers.Authorization = `Bearer ${token}`;
        }

        if (tenantId) {
            config.headers["x-tenant-id"] = tenantId;
        }

        return config;
    },
    (error) => Promise.reject(error)
);

export default api;

24. TENANT UTILITY
==================
File: frontend/src/utils/tenant.js

export const getTenantIdentifier = () => {
    const hostname = window.location.hostname.toLowerCase();

    if (
        hostname === "localhost" ||
        hostname === "127.0.0.1"
    ) {
        return null;
    }

    const parts = hostname.split(".");

    if (
        parts.length >= 3 &&
        parts.slice(-2).join(".") === "mytenantdemo.site"
    ) {
        return parts[0];
    }

    return null;
};

Example:
bb.mytenantdemo.site -> bb

25. LOGIN SERVICE
=================
File: frontend/src/features/auth/services/authService.js

import api from "../../../services/api";

export const login = async (email, password, tenantId) => {
    const response = await api.post(
        "/auth/login",
        {
            email,
            password
        },
        {
            headers: {
                "x-tenant-id": tenantId
            }
        }
    );

    return response.data;
};

IMPORTANT ARCHITECTURE NOTE
===========================
This project uses database-per-tenant architecture.

Central database:
multitenant_central

Tenant databases:
tenant_abc
tenant_bb
tenant_xyz

Each tenant database stores its own:
users
roles
permissions
employees
appointments
and other tenant-specific data.

No tenantId column is required inside tenant tables because each
tenant already has a separate database.

Prisma 7 can manage the central schema and migrations.

The existing raw pg-based tenant database connection can remain
in place for tenant isolation. Do not replace the working tenant
database layer unnecessarily.
